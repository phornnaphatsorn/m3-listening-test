let score = 0;
let progress = 0;

function speak(text) {
    if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'en-US';
        utterance.rate = 0.85; // ความเร็วสปีชระดับฟังง่ายสำหรับเด็ก
        window.speechSynthesis.speak(utterance);
    } else {
        alert("เบราว์เซอร์ของคุณไม่รองรับเสียงพูดอัตโนมัติ");
    }
}

function renderQuiz() {
    const container = document.getElementById('quiz-container');
    container.innerHTML = '';

    quizData.forEach((q, idx) => {
        if (q.section) {
            const sec = document.createElement('div');
            sec.className = 'section-title';
            sec.innerText = q.section;
            container.appendChild(sec);
        }

        const card = document.createElement('div');
        card.className = 'question-card';

        let optionsHtml = '';
        q.options.forEach((opt, optIdx) => {
            optionsHtml += `<li class="option-item" onclick="checkAnswer(${idx}, ${optIdx})">${opt}</li>`;
        });

        card.innerHTML = `
            <button class="audio-btn" onclick="speak('${q.audioText.replace(/'/g, "\\'")}')">
                🔊 กดเพื่อฟังเสียงข้อที่ ${idx + 1}
            </button>
            <div><strong>${q.question}</strong></div>
            <ul class="options-list" id="opts-${idx}">${optionsHtml}</ul>
            <div class="explanation" id="exp-${idx}">
                <div>${q.script}</div>
                <hr style="border:0; border-top:1px solid #e2e8f0; margin:8px 0;">
                <div><b>เฉลย:</b> ${q.exp}</div>
            </div>
        `;
        container.appendChild(card);
    });
}

function checkAnswer(qIdx, selectedOpt) {
    const optsContainer = document.getElementById(`opts-${qIdx}`);
    const options = optsContainer.querySelectorAll('.option-item');
    const exp = document.getElementById(`exp-${qIdx}`);
    const qData = quizData[qIdx];

    if (options[0].classList.contains('disabled')) return;

    progress++;
    document.getElementById('progress').innerText = `${progress}/15`;

    options.forEach(opt => opt.classList.add('disabled'));

    if (selectedOpt === qData.answer) {
        options[selectedOpt].classList.add('correct');
        score++;
        document.getElementById('score').innerText = score;
    } else {
        options[selectedOpt].classList.add('wrong');
        options[qData.answer].classList.add('correct');
    }

    exp.classList.add('visible');
}

window.onload = renderQuiz;
